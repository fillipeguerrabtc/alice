# IMPORTANTE: SadTalker e face-alignment são bibliotecas LEGADAS que requerem numpy 1.x
# PyTorch 2.9.x requer numpy >= 2.0, portanto usamos PyTorch 2.5.1 (última versão
# compatível com numpy 1.26.4) para manter compatibilidade enterprise.
#
# torch 2.5.1 (Oct 29, 2024) - última versão compatível com numpy 1.x
# Referência: https://pytorch.org/get-started/previous-versions/
#
# NOTA DE SEGURANÇA: CVEs do torch 2.9.x (CVE-2025-32434, etc.) são mitigadas por:
# - Container não-root (security_opt: no-new-privileges)
# - Filesystem read-only onde aplicável
# - Network isolation via Docker networks
# - Modelos carregados apenas de fontes confiáveis (HuggingFace oficial)

torchvision==0.20.1
torchaudio==2.5.1
